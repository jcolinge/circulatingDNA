# ===========================================================

# Some R code was used to generate unix commands to execute all the necessary programs considering each chromosome size
# and centromere position. This code was the following:

library(data.table)

centro <- fread("../atlas/centromere-region.txt",data.table=F)
chr.size <- fread("../atlas/pos_chromosomes.tsv",data.table=F)

# coverage & starts/ends
cat("./coverage -verbose -withstartend -infolder /data2/BIOINFO/databases/FinaleDB/cristiano-data-2019/ -fileregexp EE -outfolder cris-healthy/")

# smooth raw coverage
for (i in 1:23)
  cat(paste0("./sgfilter -sx 0.0 -i cris-healthy/chr",centro$chr[i],"_coverage.tsv -o cris-healthy/smooth-chr",centro$chr[i],"_cover_no_norm.txt -m 4 -nl 100 -nr 100 &\n"))
for (i in c(1:12,16:20,23))
  cat("sed -n '2,",centro$centro_start[i]+1," p' cris-healthy/smooth-chr",centro$chr[i],"_cover_no_norm.txt > cris-healthy/smooth-chr",centro$chr[i],"_cover_no_norm_before.txt &\n",sep="")
options(digits=12)
for (i in 1:23)
  cat("sed -n '",centro$centro_end[i]+1,",",chr.size[i,3]+1," p' cris-healthy/smooth-chr",centro$chr[i],"_cover_no_norm.txt > cris-healthy/smooth-chr",
      centro$chr[i],"_cover_no_norm_after.txt &\n",sep="")

# WPS
for (i in c(1:12,16:20,23))
  cat(paste0("./wps-score chr",centro$chr[i]," 1 ",centro$centro_start[i]," /data2/BIOINFO/databases/FinaleDB/cristiano-data-2019 EE > cris-healthy/wps-chr",
             centro$chr[i],"-before.txt &\n"))
for (i in 1:23)
  cat(paste0("./wps-score chr",centro$chr[i]," ",centro$centro_end[i]," ", chr.size[i,3]," /data2/BIOINFO/databases/FinaleDB/cristiano-data-2019 EE > cris-healthy/wps-chr",
             centro$chr[i],"-after.txt &\n"))
for (i in c(1:12,16:20,23))
  cat(paste0("./sgfilter -i cris-healthy/wps-chr",centro$chr[i],"-before.txt -o cris-healthy/smooth-wps-chr",centro$chr[i],"-before.txt -nl 50 -nr 50 -m 4 &\n"))
for (i in 1:23)
  cat(paste0("./sgfilter -i cris-healthy/wps-chr",centro$chr[i],"-after.txt -o cris-healthy/smooth-wps-chr",centro$chr[i],"-after.txt -nl 50 -nr 50 -m 4 &\n"))

# normalize & smooth coverage
for (i in c(1:12,16:20,23))
  cat(paste0("./normalize 1 ",centro$centro_start[i]," 601 cris-healthy/chr",centro$chr[i],
             "_coverage.tsv > cris-healthy/norm-chr",centro$chr[i],"-cover-before.txt &\n"))
for (i in 1:23)
  cat(paste0("./normalize ",centro$centro_end[i]," ",chr.size[i,3]," 601 cris-healthy/chr",centro$chr[i],
             "_coverage.tsv > cris-healthy/norm-chr",centro$chr[i],"-cover-after.txt &\n"))
for (i in c(1:12,16:20,23))
  cat(paste0("./sgfilter -i cris-healthy/norm-chr",centro$chr[i],"-cover-before.txt -o cris-healthy/smooth-chr",centro$chr[i],
             "-cover-before.txt -nl 100 -nr 100 -m 4 &\n"))
for (i in 1:23)
  cat(paste0("./sgfilter -i cris-healthy/norm-chr",centro$chr[i],"-cover-after.txt -o cris-healthy/smooth-chr",centro$chr[i],
             "-cover-after.txt -nl 100 -nr 100 -m 4 &\n"))

# normalize & smooth starts/ends
for (i in c(1:12,16:20,23))
  cat(paste0("./normalize 1 ",centro$centro_start[i]," 601 cris-healthy/chr",centro$chr[i],
             "_starts.tsv > cris-healthy/norm-chr",centro$chr[i],"-starts-before.txt &\n"))
for (i in 1:23)
  cat(paste0("./normalize ",centro$centro_end[i]," ",chr.size[i,3]," 601 cris-healthy/chr",centro$chr[i],
             "_starts.tsv > cris-healthy/norm-chr",centro$chr[i],"-starts-after.txt &\n"))
for (i in c(1:12,16:20,23))
  cat(paste0("./normalize 1 ",centro$centro_start[i]," 601 cris-healthy/chr",centro$chr[i],
             "_ends.tsv > cris-healthy/norm-chr",centro$chr[i],"-ends-before.txt &\n"))
for (i in 1:23)
  cat(paste0("./normalize ",centro$centro_end[i]," ",chr.size[i,3]," 601 cris-healthy/chr",centro$chr[i],
             "_ends.tsv > cris-healthy/norm-chr",centro$chr[i],"-ends-after.txt &\n"))
for (i in c(1:12,16:20,23))
  cat(paste0("./sgfilter -i cris-healthy/norm-chr",centro$chr[i],"-starts-before.txt -o cris-healthy/smooth-chr",centro$chr[i],
             "-starts-before.txt -nl 200 -nr 200 -m 4 &\n"))
for (i in 1:23)
  cat(paste0("./sgfilter -i cris-healthy/norm-chr",centro$chr[i],"-starts-after.txt -o cris-healthy/smooth-chr",centro$chr[i],
             "-starts-after.txt -nl 200 -nr 200 -m 4 &\n"))
for (i in c(1:12,16:20,23))
  cat(paste0("./sgfilter -i cris-healthy/norm-chr",centro$chr[i],"-ends-before.txt -o cris-healthy/smooth-chr",centro$chr[i],
             "-ends-before.txt -nl 200 -nr 200 -m 4 &\n"))
for (i in 1:23)
  cat(paste0("./sgfilter -i cris-healthy/norm-chr",centro$chr[i],"-ends-after.txt -o cris-healthy/smooth-chr",centro$chr[i],
             "-ends-after.txt -nl 200 -nr 200 -m 4 &\n"))

# WPS peak detection
for (i in c(1:12,16:20,23))
  cat(paste0("./findChromatosomes -verbose -sc cris-healthy/smooth-chr",centro$chr[i],
             "-cover-before.txt -sw cris-healthy/smooth-wps-chr",centro$chr[i],
             "-before.txt -ss cris-healthy/smooth-chr",centro$chr[i],"-starts-before.txt -se cris-healthy/smooth-chr",
             centro$chr[i],"-ends-before.txt -basename cris-healthy-indiv-selections/chr",centro$chr[i],"_before -rsc cris-healthy/smooth-chr",
             centro$chr[i],"_cover_no_norm_before.txt &\n"))
for (i in 1:23)
  cat(paste0("./findChromatosomes -verbose -sc cris-healthy/smooth-chr",centro$chr[i],
             "-cover-after.txt -sw cris-healthy/smooth-wps-chr",centro$chr[i],"-after.txt -ss cris-healthy/smooth-chr",
             centro$chr[i],"-starts-after.txt -se cris-healthy/smooth-chr",centro$chr[i],
             "-ends-after.txt -basename cris-healthy-indiv-selections/chr",centro$chr[i],"_after -rsc cris-healthy/smooth-chr",
             centro$chr[i],"_cover_no_norm_after.txt &\n"))

# A bash script was used to preprocess individual patient files, extract 167bp fragments from each file, merge all 167bp fragments and extract k-stacked fragments.
# This code was the following

#!/bin/bash

# list all individual patient files
cd cristiano-data-2019
for f in *.tsv;
do
        list_files1+=($f)
done

# for each file extract 167bp fragments with quality score > 30, and keep only those on chromosomes 1 to 22 and chromosomes X,Y (no contigs)
for ((f=0; f<${#list_files1[@]}; f++))
do
	echo "${list_files1[f]}"
        awk -v len=167 'BEGIN {OFS="\t"} ($3 - $2 == len) && ($4 > 30) {print}' "${list_files1[f]}" > "cris-output/${list_files1[f]%.*}_167_frags.bed"
	awk 'BEGIN{OFS="\t"} {sub(/^chr/, "", $1)} ($1 ~ /^[1-9][0-9]*$|^X$|^Y$/) {print}' "cris-output/${list_files1[f]%.*}_167_frags.bed" > "cris-output/${list_files1[f]%.*}_167_frags_2.bed"
	rm "cris-output/${list_files1[f]%.*}_167_frags.bed"
done

# concatenate all 167bp fragments into one file
cd cris-output
bedops --everything *167_frags_2.bed* > concat_all_167_score.bed

# remove fragments at centromeres and on chromosome Y and sort them
bedtools intersect -a concat_all_167_score.bed -b centromere-region.bed -v > concat_all_167_score_no_centro.bed
awk 'BEGIN{OFS="\t"} $1!="Y" {print}' concat_all_167_score_no_centro.bed > concat_all_167_score_no_centro_noY.bed
sort -k1,1 -k2,2n concat_all_167_score_no_centro_noY.bed > concat_all_167_score_no_centro_noY_sort.bed

# count fragments with identical positions
bedtools intersect -a concat_all_167_score_no_centro_noY_sort.bed -b concat_all_167_score_no_centro_noY_sort.bed -sorted -f 0.999 -c > overlapping_all_167.bed

# create files with k-stacked fragments with unique coordinates for k in 1 to 20, and retrieve their sequences from reference genome hg38
for i in $(seq 1 20);
do
python merge_frags_empils.py overlapping_all_167 unique_nucleos_all_167_"$i"x.bed $i
bedtools getfasta -fi Homo_sapiens.GRCh38.dna.primary_assembly.fa -bed unique_nucleos_all_267_"$i"x.bed > unique_nucleos_all_267_"$i"x.fa.out
done
